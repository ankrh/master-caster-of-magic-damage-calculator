    int16_t damage_types[NUM_DAMAGE_TYPES], int16_t attack_strength_override)

            running_front_damage = 0;           /* 131:0x87279  160:=  com1:= */

                damage = running_front_damage;  /* 131:0x8728D  160:=  com1:= */
                running_front_damage = 0;       /* 131:0x87290  160:=  com1:= */

                damage = 0;                     /* 131:0x87297  160:=  com1:= */

                                      /* JNE -> loop test 131:0x872D6..0x872DB  160:=  com1:= */
                    damage -= CMB_DefenseRoll(defense_special, defender_to_block);
                                      /* far 03D0:0025; 131:0x872DD..0x872EA  160:=  com1:= */

                                      /* raw high-word mask 0x8000; 131:0x872EC..0x872FA  160:=  com1:= */
                        damage -= 2;  /* 131:0x872FC  160:=  com1:= */

                damage = 0;           /* 131:0x87319  160:=  com1:= */

        damage_total += damage + figures_lost * (int8_t)bu->hits;
                                      /* 131:0x8731B..0x87334  160:=  com1:= */
        damage = 0;                   /* SI zero 131:0x87337  160:=  com1:= */
        figures_lost = 0;             /* 131:0x87339  160:=  com1:= */

### E2 — `BU_ApplyDamage` `[0x87389,0x876C0)`

```c
void __far BU_ApplyDamage(int16_t battle_unit_idx,
                          int16_t damage_types[NUM_DAMAGE_TYPES])
{
    struct s_BATTLE_UNIT __far *bu = &_battle_units[battle_unit_idx];
    int16_t damage_total = 0;         /* 131:0x87394  160:=  com1:= */
    for (int16_t itr = 0; itr < NUM_DAMAGE_TYPES; ++itr)
                                      /* 131:0x87399..0x873AF  160:=  com1:= */
        damage_total += damage_types[itr];
                                      /* 131:0x8739D..0x873A8  160:=  com1:= */

    if (damage_total <= 0)            /* JG -> status test 131:0x873B1..0x873B5  160:=  com1:= */
        return;                       /* JMP epilogue 131:0x873B7  160:=  com1:= */
    if ((int8_t)bu->status != BUS_ACTIVE)
                                      /* JE -> bucket loop 131:0x873BA..0x873CC  160:=  com1:= */
        return;                       /* JMP epilogue 131:0x873CE  160:=  com1:= */

    for (int16_t itr = 0; itr < NUM_DAMAGE_TYPES; ++itr) {
                                      /* 131:0x873D1..0x87447  160:=  com1:= */
        if ((uint8_t)bu->damage[itr] + damage_types[itr] > DAMAGE_BUCKET_CAP)
                                      /* word compare to raw 0x00C8; 131:0x873D5..0x873F8  160:=  com1:= */
            bu->damage[itr] = DAMAGE_BUCKET_CAP;
                                      /* byte state write raw 0xC8; 131:0x873FA..0x87409  160:=  com1:= */
        else
            bu->damage[itr] += (uint8_t)damage_types[itr];
                                      /* byte state write; 131:0x87410..0x8743F  160:=  com1:= */
    }

    damage_total += (int8_t)bu->front_figure_damage;
                                      /* 131:0x87449..0x8745B  160:=  com1:= */
    if (damage_total > 0) {           /* JG -> division 131:0x8745E..0x87462  160:=  com1:= */
        int16_t figures_lost = damage_total / (int8_t)bu->hits;
                                      /* full CWD/IDIV idiom 131:0x87467..0x87481  160:=  com1:= */
        if ((int8_t)bu->Cur_Figures < figures_lost)
                                      /* signed JGE -> subtraction 131:0x87484..0x87499  160:=  com1:= */
            figures_lost = (int8_t)bu->Cur_Figures;
                                      /* 131:0x8749B..0x874AD  160:=  com1:= */
        bu->Cur_Figures -= (uint8_t)figures_lost;
                                      /* byte state write 131:0x874B0..0x874D3  160:=  com1:= */
        bu->front_figure_damage = damage_total % (int8_t)bu->hits;
                                      /* CWD/IDIV remainder; byte state write 131:0x874D7..0x87500  160:=  com1:= */
    }

    if ((int8_t)bu->Cur_Figures <= 0) {
                                      /* JLE -> death block 131:0x87504..0x87516  160:=  com1:= */
#if BUILD == BUILD_CP160 || BUILD == BUILD_COM1
        /* 160:0x8751B..0x87522 and com1:0x8751B..0x87522 are eight NOPs. */
        bu->status = BUS_DEAD;        /* raw 4; state write 131:—  160:0x87523  com1:0x87523 */
#endif
        bu->Cur_Figures = 0;          /* byte state write 131:0x8751B..0x87528
                                         160:0x87528  com1:0x87528 */
        _combat_winner = Eliminated_Opponent();
                                      /* push-cs near far-call + global DS:0xC972 write
                                         131:0x8752E..0x87532  160:=  com1:= */
        int16_t is_summoned = Battle_Unit_Is_Summoned_Creature(battle_unit_idx);
                                      /* far 03E0:005C; 131:0x87535..0x8753B  160:=  com1:= */
#if BUILD == BUILD_MOM131 || BUILD == BUILD_CP160
        if (is_summoned != ST_FALSE)  /* JE -> damage classification 131:0x8753C..0x8753E  160:= */
            _UNITS[bu->unit_idx].wp = UNIT_WP_DEAD_COMBAT_SUMMON;
                                      /* raw 9; persistent state write 131:0x87540..0x8755B  160:= */
#else
        (void)is_summoned;            /* AX result has no reader; com1:0x8753C begins next push */
        overlay_03D0_004D(UNKNOWN_C520);
                                      /* pushes DS:0xC520; far 03D0:004D; com1:0x8753C..0x87545 */
        /* com1:0x87546..0x8755F are 26 NOPs. */
#endif

        if ((uint8_t)bu->damage[DAMAGE_IRREVERSIBLE]
                >= (uint8_t)bu->damage[DAMAGE_UNDEATH]
                                      /* unsigned JB -> next class 131:0x87560..0x87584  160:=  com1:= */
            && (uint8_t)bu->damage[DAMAGE_IRREVERSIBLE]
                >= (uint8_t)bu->damage[DAMAGE_REGULAR]) {
                                      /* unsigned JB -> next class 131:0x87586..0x875AA  160:=  com1:= */
            bu->status = BUS_GONE;    /* raw 6; state write 131:0x875AC..0x875B9  160:=  com1:= */
        } else if ((uint8_t)bu->damage[DAMAGE_UNDEATH]
                       > (uint8_t)bu->damage[DAMAGE_IRREVERSIBLE]
                                      /* unsigned JBE -> regular class 131:0x875C1..0x875E5  160:=  com1:= */
                   && (uint8_t)bu->damage[DAMAGE_UNDEATH]
                       >= (uint8_t)bu->damage[DAMAGE_REGULAR]) {
                                      /* unsigned JB -> regular class 131:0x875E7..0x8760B  160:=  com1:= */
            if (_UNITS[bu->unit_idx].wp == UNIT_WP_DEAD_COMBAT_SUMMON)
                                      /* raw 9; JNE -> drained 131:0x8760D..0x8762D  160:=  com1:= */
                bu->status = BUS_GONE;/* raw 6; state write 131:0x8762F..0x8763C  160:=  com1:= */
            else
                bu->status = BUS_DRAINED;
                                      /* raw 5; state write 131:0x87643..0x87650  160:=  com1:= */
        } else if ((uint8_t)bu->damage[DAMAGE_REGULAR]
                       > (uint8_t)bu->damage[DAMAGE_IRREVERSIBLE]
                                      /* unsigned JBE -> update call 131:0x87657..0x8767B  160:=  com1:= */
                   && (uint8_t)bu->damage[DAMAGE_REGULAR]
                       > (uint8_t)bu->damage[DAMAGE_UNDEATH]) {
                                      /* unsigned JBE -> update call 131:0x8767D..0x876A1  160:=  com1:= */
            bu->status = BUS_DEAD;    /* raw 4; state write 131:0x876A3..0x876B0  160:=  com1:= */
        }

        overlay_0348_003E();          /* no pushed argument and no consumed result;
                                         far 0348:003E; 131:0x876B5  160:=  com1:= */
    }
}                                     /* epilogue/retf 131:0x876BA..0x876BF  160:=  com1:= */
```

The CP/CoM `BUS_DEAD` pre-write uses the still-valid battle-unit pointer left in `ES:BX`: bytes
`26 C6 47 34 04` at `0x87523`. CoM 1's opaque call is bytes `9A 4D 00 D0 03` at `0x87540`.

                          | _battle_units[defender_idx].item_enchantments

                                      /* 131:0x87729..0x87781  160:=  com1:= */

                                      /* high stored word raw 0x0040; JNE -> allow 131:0x87789..0x87797  160:=  com1:= */
        || (_battle_units[attacker_idx].Attribs_1 & USA_IMMUNITY_ILLUSION)) {
                                      /* raw 0x0008; JE -> visibility tests 131:0x87799..0x877AC  160:=  com1:= */

                                      /* low stored word raw 0x8000; JNE -> blocked 131:0x877B5..0x877C4  160:=  com1:= */
               || (_battle_units[defender_idx].Abilities & UA_INVISIBILITY)) {
                                      /* raw 0x0040; JE -> darkness gate 131:0x877C6..0x877D9  160:=  com1:= */
        result = 1;                   /* raw 1; 131:0x877DB  160:=  com1:= */
    }

    if (battlefield->wall_of_darkness == ST_TRUE) {
                                      /* field +0x157A, raw 1; 131:0x877E0..0x877EA
                                         160:0x877E0..0x877EA  com1:= */

        if (!(attacker_all & UE_TRUE_SIGHT)) {
                                      /* raw high-word mask 0x0040; JNE -> return 131:0x877EC..0x877FA */

## Merged findings

### Spell damage is a complete figure-by-figure combat attack, not a flat hit-point subtraction

`Apply_Battle_Unit_Damage_From_Spell` first clears all three caller damage buckets
(`0x870A8..0x870C4`), then rejects Chaos/Death spells against Righteousness
(`0x870C6..0x87117`) and every spell against Magic Immunity (`0x87100..0x87117`). It feeds the
spell's immunity byte, attack flags and realm into `Battle_Unit_Defense_Special` at `0x871BB`, using
ranged-type literal 38 in both MoM builds and 39 in CoM 1 (`0x871B6`). The remaining attack uses
`CMB_AttackRoll` and `CMB_DefenseRoll` (`0x87239`, `0x87248`, `0x872E3`), subtracts two for
Invulnerability after each defense roll (`0x87261`, `0x872FC`), and carries damage across figures.

Area spells set the strike count to the defender's current figure count and add Damage Limit
(`0x871C6..0x871E8`). Warp Lightning instead starts with one strike per point of strength and
decrements strength after each strike (`0x871ED..0x871F7`, `0x8733E..0x87345`). Black Sleep adds
the automatic/Doom-damage flag (`0x87201..0x8721C`), bypassing both attack and defense rolls.

### `BU_ApplyDamage` rejects inert inputs before changing any unit state

The routine returns immediately when the signed sum of the three input buckets is nonpositive
(`0x873B1..0x873B7`) or the target status is not Active (`0x873BA..0x873CE`). Only then does it add
each bucket to the corresponding byte counter, saturating each at 200 (`0x873D1..0x87447`). It
uses signed `CWD / IDIV` for figures lost and the front-figure remainder
(`0x87467..0x87500`), caps figures lost at `Cur_Figures`, and clamps the final figure count to zero.

After the last figure dies, irreversible damage wins ties with both other buckets
(`0x87560..0x875BE`). Undeath damage wins only when it strictly exceeds irreversible damage and is
at least regular damage (`0x875C1..0x87655`); regular damage wins only when it strictly exceeds both
(`0x87657..0x876B0`). The corresponding statuses are Gone, Drained/Gone, and Dead. MoM 1.31 and CP
1.60 mark a recognized combat summon's persistent unit `wp` byte as 9 (`0x87536..0x8755B`). CoM 1
still calls the predicate but discards its result, calls opaque `03D0:004D` with the word at
`DS:0xC520`, and replaces the persistent write with 26 NOPs (`0x87536..0x8755F`). CP 1.60 and CoM
1 also pre-write status Dead at `0x87523`; 1.31 recomputes the unit pointer there instead.

### Ranged visibility and Wall of Darkness use all three enchantment sources

`Check_Attack_Ranged` ORs permanent, live battle and item enchantments for both units
(`0x876CE..0x87781`). Attacker True Sight or Illusions Immunity makes the initial visibility result
clear; otherwise defender Invisibility from either enchantments or abilities returns the invisible
classification (`0x87784..0x877DB`). Wall of Darkness can then override that result only when the
opaque defender helper returns 1 and the attacker helper returns 0. MoM 1.31 exempts attacker True
Sight (`0x877EC..0x877FA`); CP 1.60 and CoM 1 instead test the attacker battle unit's byte-wide
Illusions Immunity bit (`0x877F7..0x877FC`).

### Elimination counts confused units on the side they currently fight for

`Eliminated_Opponent` ignores non-Active units (`0x8847D..0x8848F`). An attacker-controlled unit
whose Confusion state is 2 increments the defender tally, and a defender-controlled unit in that
state increments the attacker tally (`0x88491..0x884F2`). A zero attacker tally returns the combat
defender; otherwise a zero defender tally returns the attacker; otherwise it returns `-1`
(`0x884FA..0x8850C`). Because the attacker tally is tested first, a battle with neither side
counted returns the defender player.

### The city-wall cell predicate excludes the inner two-by-two cells

`Combat_Grid_Cell_Has_City_Wall` first requires a walled battlefield and the inclusive combat-grid
box `x=5..8`, `y=10..13` (`0x9F053..0x9F071`). It excludes the four cells where
`x in {6,7}` and `y in {11,12}` (`0x9F073..0x9F085`), then indexes the `int16_t walls[4][4]`
array with an eight-byte row stride and two-byte column stride (`0x9F087..0x9F0A5`). The routine is
byte-identical in all three builds.

### The combat-summon predicate recognizes five spell-defined unit types plus Demon

`Battle_Unit_Is_Summoned_Creature` compares the unit's zero-extended type byte against the
`unit_type` words in the Fire Elemental, Earth Elemental, Phantom Beast, Phantom Warriors and Air
Elemental spell records, then against raw unit type `0xA9` (Demon), at `0x9F0DA..0x9F122`. Any
match returns true through the shared funnel at `0x9F0E5`; the routine is byte-identical in all
three builds.
